#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
栖梧7班班级管理系统 · 校园网同步服务端（Python 版 · R24 每设备数据槽）

零第三方依赖：只用 Python 标准库。Windows / Linux / macOS 通用。
Python 3.6+ 即可（Windows 上装 Python 时记得勾选 "Add Python to PATH"）。

做什么：
  1) 托管主程序网页（浏览器打开 http://服务器IP:端口 就能用）
  2) 提供 /api/sync 接口，把每个设备的数据存成独立「数据槽」
     —— 教师端 / 班级端 / 个人端各写各的槽，读的时候把所有槽合并，
       任何一台设备都不可能覆盖别人刚写的数据（R24 之前是整包替换，
       学生端一同步就可能把教师端推的名单/成绩冲掉，现已根治）

数据文件：本脚本同目录 qiwu7_data/data.json（自动创建，自动留 10 份历史备份）
"""
import json, os, sys, time, shutil, threading
from http.server import ThreadingHTTPServer, SimpleHTTPRequestHandler
from urllib.parse import urlparse, parse_qs, quote

# ============ 配置 ============
PORT = int(sys.argv[1]) if len(sys.argv) > 1 else int(os.environ.get('QIWU7_PORT', '8080'))
TOKEN = os.environ.get('QIWU7_TOKEN', 'qiwu7')      # 班级口令，务必改成自己的
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_DIR = os.path.join(BASE_DIR, 'qiwu7_data')
DATA_FILE = os.path.join(DATA_DIR, 'data.json')
BACKUP_DIR = os.path.join(DATA_DIR, 'backups')
MAX_BACKUPS = 10
SLOT_IDLE_PRUNE_MS = 180 * 24 * 3600 * 1000   # 数据槽闲置超过 180 天才允许清理（避免暑假误删）
LOCK = threading.Lock()

os.makedirs(DATA_DIR, exist_ok=True)
os.makedirs(BACKUP_DIR, exist_ok=True)


def log(msg):
    print('[%s] %s' % (time.strftime('%H:%M:%S'), msg), flush=True)


def empty_slots():
    return {'rev': 0, 'ts': 0, 'slots': {}}


def migrate(d):
    """旧版（整包 payload）数据文件 → 每设备数据槽结构；迁移是幂等的"""
    if not isinstance(d, dict):
        return empty_slots()
    if isinstance(d.get('slots'), dict):
        d.setdefault('rev', 0)
        d.setdefault('ts', 0)
        return d
    if d.get('payload') is not None:
        # R23 及更早的整包存档：包成名为 legacy 的槽，新客户端合并时会自然吸收
        ts = d.get('ts') or int(time.time() * 1000)
        return {'rev': d.get('rev', 0), 'ts': ts,
                'slots': {'legacy': {'rev': d.get('rev', 0), 'ts': ts, 'payload': d['payload']}}}
    return empty_slots()


def read_data():
    """读取服务器上的数据；没有就返回空包（rev=0）。旧格式自动迁移。"""
    if not os.path.exists(DATA_FILE):
        return empty_slots()
    try:
        with open(DATA_FILE, 'r', encoding='utf-8') as f:
            d = json.load(f)
        return migrate(d)
    except Exception as e:
        log('数据文件损坏：%s —— 自动回退到最近一份备份' % e)
        restore_latest_backup()
        return empty_slots()


def backup_current():
    """写入前把现有文件复制一份到 backups/，只保留最近 MAX_BACKUPS 份"""
    if not os.path.exists(DATA_FILE):
        return
    name = 'data-%s.json' % time.strftime('%Y%m%d-%H%M%S')
    try:
        shutil.copy2(DATA_FILE, os.path.join(BACKUP_DIR, name))
    except Exception as e:
        log('备份失败：%s' % e)
        return
    try:
        bks = sorted([f for f in os.listdir(BACKUP_DIR) if f.startswith('data-')])
        while len(bks) > MAX_BACKUPS:
            os.remove(os.path.join(BACKUP_DIR, bks.pop(0)))
    except Exception:
        pass


def restore_latest_backup():
    try:
        bks = sorted([f for f in os.listdir(BACKUP_DIR) if f.startswith('data-')])
        if not bks:
            return False
        shutil.copy2(os.path.join(BACKUP_DIR, bks[-1]), DATA_FILE)
        log('已恢复备份：%s' % bks[-1])
        return True
    except Exception:
        return False


def write_file(d):
    """原子落盘：先写临时文件再替换，断电也不会写坏"""
    backup_current()
    d['ts'] = int(time.time() * 1000)
    tmp = DATA_FILE + '.tmp'
    with open(tmp, 'w', encoding='utf-8') as f:
        json.dump(d, f, ensure_ascii=False)
    os.replace(tmp, DATA_FILE)


def prune_slots(d, keep_dev=None):
    """清理闲置超过 180 天的数据槽（刚写入的槽除外；只剩一个槽时不清）"""
    now = int(time.time() * 1000)
    slots = d.get('slots') or {}
    stale = [dev for dev, s in slots.items()
             if dev != keep_dev and now - (s.get('ts') or 0) > SLOT_IDLE_PRUNE_MS]
    if stale and len(slots) - len(stale) >= 1:
        for dev in stale:
            del slots[dev]
        log('已清理 %d 个闲置数据槽：%s' % (len(stale), ', '.join(stale)))


def write_slot(payload, dev, base_rev):
    """
    写入某个设备自己的数据槽（槽级乐观锁）：
      - base_rev 与该槽当前 rev 不一致 → 拒绝（409），客户端重新拉取合并后再试
      - 一致 → 写入，该槽 rev+1，全局 rev+1
    各设备写各的槽，互相之间永远不会冲突。
    """
    with LOCK:
        d = read_data()
        slots = d.setdefault('slots', {})
        cur = slots.get(dev) or {'rev': 0, 'ts': 0, 'payload': None}
        if base_rev is not None and cur.get('rev', 0) != base_rev:
            return {'ok': False, 'conflict': True, 'rev': d.get('rev', 0),
                    'slotRev': cur.get('rev', 0), 'payload': cur.get('payload')}
        d['rev'] = d.get('rev', 0) + 1
        slots[dev] = {'rev': d['rev'], 'ts': int(time.time() * 1000), 'payload': payload}
        prune_slots(d, keep_dev=dev)
        write_file(d)
        return {'ok': True, 'rev': d['rev'], 'ts': d['ts'], 'slotRev': slots[dev]['rev']}


def write_legacy(payload, base_rev):
    """兼容 R23 旧客户端：整包替换（保留旧语义，便于混合升级期间过渡）"""
    with LOCK:
        d = read_data()
        if base_rev is not None and d.get('rev', 0) != base_rev:
            return {'ok': False, 'conflict': True, 'rev': d.get('rev', 0),
                    'payload': {'slots': d.get('slots')}}
        d['rev'] = d.get('rev', 0) + 1
        d['slots'] = {'legacy': {'rev': d['rev'], 'ts': int(time.time() * 1000), 'payload': payload}}
        write_file(d)
        return {'ok': True, 'rev': d['rev'], 'ts': d['ts']}


class Handler(SimpleHTTPRequestHandler):
    """静态文件 + 同步接口"""

    def _send_json(self, obj, code=200):
        body = json.dumps(obj, ensure_ascii=False).encode('utf-8')
        self.send_response(code)
        self.send_header('Content-Type', 'application/json; charset=utf-8')
        self.send_header('Content-Length', str(len(body)))
        self.send_header('Cache-Control', 'no-store')
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.send_header('Access-Control-Allow-Methods', 'GET,POST,OPTIONS')
        self.end_headers()
        self.wfile.write(body)

    def _check_token(self, q):
        return q.get('token', [''])[0] == TOKEN

    def do_OPTIONS(self):
        self._send_json({'ok': True})

    def do_GET(self):
        u = urlparse(self.path)
        q = parse_qs(u.query)

        if u.path == '/api/ping':
            return self._send_json({'ok': True, 'name': 'qiwu7', 'rev': read_data()['rev'],
                                    'time': time.strftime('%Y-%m-%d %H:%M:%S')})

        if u.path == '/api/sync':
            if not self._check_token(q):
                return self._send_json({'ok': False, 'error': '口令不对'}, 401)
            d = read_data()
            slots = d.get('slots') or {}
            # 客户端带上各槽已知版本号（revs={设备id:版本}）→ 只回传有变化的槽，省带宽
            revs = {}
            raw_revs = q.get('revs', [''])[0]
            if raw_revs:
                try:
                    revs = json.loads(raw_revs)
                    if not isinstance(revs, dict):
                        revs = {}
                except Exception:
                    revs = {}
            if revs:
                changed = {dev: s for dev, s in slots.items()
                           if not (dev in revs and revs[dev] == s.get('rev'))}
            else:
                changed = slots
            return self._send_json({'ok': True, 'rev': d['rev'], 'ts': d['ts'], 'slots': changed,
                                    'count': len(slots),
                                    'slotsRev': {dev: s.get('rev', 0) for dev, s in slots.items()}})

        if u.path == '/api/backup':
            if not self._check_token(q):
                return self._send_json({'ok': False, 'error': '口令不对'}, 401)
            d = read_data()
            body = json.dumps(d, ensure_ascii=False).encode('utf-8')
            fn = '栖梧7班数据备份-%s.json' % time.strftime('%Y%m%d-%H%M')
            self.send_response(200)
            self.send_header('Content-Type', 'application/json; charset=utf-8')
            # HTTP 头只能装 ASCII：中文文件名必须按 RFC 5987 百分号编码，否则 send_header 抛 UnicodeEncodeError
            self.send_header('Content-Disposition', 'attachment; filename*=UTF-8\'\'%s' % quote(fn))
            self.send_header('Content-Length', str(len(body)))
            self.end_headers()
            self.wfile.write(body)
            return

        # 其余走静态文件
        return SimpleHTTPRequestHandler.do_GET(self)

    def do_POST(self):
        u = urlparse(self.path)
        q = parse_qs(u.query)

        if u.path != '/api/sync':
            return self._send_json({'ok': False, 'error': 'not found'}, 404)

        if not self._check_token(q):
            return self._send_json({'ok': False, 'error': '口令不对'}, 401)

        try:
            length = int(self.headers.get('Content-Length') or 0)
        except ValueError:
            length = 0
        raw = self.rfile.read(length).decode('utf-8') if length else ''
        try:
            req = json.loads(raw) if raw else {}
        except Exception as e:
            return self._send_json({'ok': False, 'error': '请求不是合法 JSON：%s' % e}, 400)

        payload = req.get('payload')
        base_rev = req.get('baseRev')
        if payload is None:
            return self._send_json({'ok': False, 'error': '缺少 payload'}, 400)

        dev = req.get('deviceId')
        if dev is not None:
            # R24 新协议：写自己的数据槽
            dev = str(dev)[:64] or 'unknown'
            r = write_slot(payload, dev, base_rev)
        else:
            # 旧协议：整包替换（兼容升级过渡期的旧客户端）
            r = write_legacy(payload, base_rev)

        if not r.get('ok'):
            # 冲突：把服务器最新版本一起返回，客户端合并后重试
            return self._send_json({'ok': False, 'conflict': True, 'rev': r.get('rev', 0),
                                    'slotRev': r.get('slotRev'),
                                    'payload': r.get('payload'),
                                    'slots': r.get('payload', {}).get('slots') if isinstance(r.get('payload'), dict) else None,
                                    'error': '数据已被其他设备更新，请合并后重试'}, 409)
        log('已同步 rev=%s（%s，来自 %s）' % (r['rev'], dev or 'legacy', self.client_address[0]))
        return self._send_json({'ok': True, 'rev': r['rev'], 'ts': r['ts'], 'slotRev': r.get('slotRev')})

    def log_message(self, fmt, *args):
        # 只在出错时打印，避免刷屏
        if '404' in (fmt % args) or '500' in (fmt % args):
            log(fmt % args)


def main():
    # 静态根目录 = 本脚本所在目录
    os.chdir(BASE_DIR)
    try:
        srv = ThreadingHTTPServer(('0.0.0.0', PORT), Handler)
    except OSError as e:
        print('端口 %s 起不来：%s\n换一个端口试试，例如：python qiwu7_server.py 9000' % (PORT, e))
        sys.exit(1)

    import socket
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        s.connect(('8.8.8.8', 80))
        ip = s.getsockname()[0]
        s.close()
    except Exception:
        ip = '127.0.0.1'

    print('=' * 62)
    print('  栖梧7班班级管理系统 · 校园网服务端已启动（R24 数据槽版）')
    print('=' * 62)
    print('  本机访问：   http://127.0.0.1:%s' % PORT)
    print('  局域网访问： http://%s:%s   ← 把这个地址发给老师 / 输进一体机' % (ip, PORT))
    print('  班级口令：   %s' % TOKEN)
    if TOKEN == 'qiwu7':
        print('  ' + '!' * 58)
        print('  !! 安全警告：你现在用的是【默认口令 qiwu7】')
        print('  !! 这个口令是公开的（写在启动脚本和说明文档里），')
        print('  !! 校园网内任何知道本机地址的人都能下载全班数据、')
        print('  !! 给自己加分，甚至覆盖全班记录。')
        print('  !! 强烈建议改掉（改完要重启本服务）：')
        print('  !!   Windows: 改 启动_Windows.bat 第 13 行 set QIWU7_TOKEN=你的口令')
        print('  !!   Linux:   QIWU7_TOKEN=你的口令 ./启动_Linux.sh')
        print('  ' + '!' * 58)
    print('  数据文件：   %s' % DATA_FILE)
    print('  停止服务：   按 Ctrl + C（systemd 服务用 sudo systemctl stop qiwu7）')
    print('=' * 62)
    print('  提醒：第一次打开网页后，要去「设置 / 数据 → 🌐 校园网同步」')
    print('        填上面的地址与口令，点「立即同步」，三端就通了。')
    print('=' * 62)
    sys.stdout.flush()          # 立刻显示，避免被输出缓冲吞掉
    try:
        srv.serve_forever()
    except KeyboardInterrupt:
        print('\n已停止。')
        srv.server_close()


if __name__ == '__main__':
    main()

#!/usr/bin/env node
/**
 * 栖梧7班班级管理系统 · 校园网同步服务端（Node 版 · R24 每设备数据槽）
 *
 * 零第三方依赖：只用 Node 自带模块。装了 Node 就能跑（node -v 查看有没有）。
 * 与 Python 版功能完全一致（R24 分槽存储），二选一即可 —— 推荐优先用 Python 版（更好装）。
 *
 * 用法：
 *   node qiwu7_server.js            // 默认 8080
 *   node qiwu7_server.js 9000       // 指定端口
 *   set QIWU7_TOKEN=xxxx && node qiwu7_server.js    // Windows 设口令
 *   QIWU7_TOKEN=xxxx node qiwu7_server.js           // Linux 设口令
 *
 * 数据文件：本脚本同目录 qiwu7_data/data.json（自动创建，自动留 10 份历史备份）
 */
'use strict';
const http = require('http');
const fs = require('fs');
const path = require('path');

const PORT = parseInt(process.argv[2], 10) || parseInt(process.env.QIWU7_PORT || '8080', 10);
const TOKEN = process.env.QIWU7_TOKEN || 'qiwu7';
const BASE_DIR = __dirname;
const DATA_DIR = path.join(BASE_DIR, 'qiwu7_data');
const DATA_FILE = path.join(DATA_DIR, 'data.json');
const BACKUP_DIR = path.join(DATA_DIR, 'backups');
const MAX_BACKUPS = 10;
const SLOT_IDLE_PRUNE_MS = 180 * 24 * 3600 * 1000;   // 数据槽闲置 180 天才允许清理
const INDEX = path.join(BASE_DIR, 'index.html');   // 主程序（复制一份改名 index.html')

if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true });
if (!fs.existsSync(BACKUP_DIR)) fs.mkdirSync(BACKUP_DIR, { recursive: true });

const log = m => console.log(`[${new Date().toTimeString().slice(0, 8)}] ${m}`);

function emptyData() { return { rev: 0, ts: 0, slots: {} }; }

function migrate(d) {
  /* 旧版（整包 payload）→ 每设备数据槽；幂等 */
  if (!d || typeof d !== 'object') return emptyData();
  if (d.slots && typeof d.slots === 'object') { d.rev = d.rev || 0; d.ts = d.ts || 0; return d; }
  if (d.payload !== undefined) {
    const ts = d.ts || Date.now();
    return { rev: d.rev || 0, ts, slots: { legacy: { rev: d.rev || 0, ts, payload: d.payload } } };
  }
  return emptyData();
}

function readData() {
  if (!fs.existsSync(DATA_FILE)) return emptyData();
  try {
    return migrate(JSON.parse(fs.readFileSync(DATA_FILE, 'utf8')));
  } catch (e) {
    log('数据文件损坏：' + e.message + ' —— 自动回退到最近一份备份');
    restoreLatest();
    return emptyData();
  }
}
function backupCurrent() {
  if (!fs.existsSync(DATA_FILE)) return;
  try {
    fs.copyFileSync(DATA_FILE, path.join(BACKUP_DIR, `data-${new Date().toISOString().replace(/[:.]/g, '-')}.json`));
    const bks = fs.readdirSync(BACKUP_DIR).filter(f => f.startsWith('data-')).sort();
    while (bks.length > MAX_BACKUPS) fs.unlinkSync(path.join(BACKUP_DIR, bks.shift()));
  } catch (e) { log('备份失败：' + e.message); }
}
function restoreLatest() {
  try {
    const bks = fs.readdirSync(BACKUP_DIR).filter(f => f.startsWith('data-')).sort();
    if (!bks.length) return;
    fs.copyFileSync(path.join(BACKUP_DIR, bks[bks.length - 1]), DATA_FILE);
    log('已恢复备份：' + bks[bks.length - 1]);
  } catch (e) { /* ignore */ }
}
function writeFile(d) {
  /* 原子落盘：先写临时文件再替换 */
  backupCurrent();
  d.ts = Date.now();
  const tmp = DATA_FILE + '.tmp';
  fs.writeFileSync(tmp, JSON.stringify(d), 'utf8');
  fs.renameSync(tmp, DATA_FILE);
}
function pruneSlots(d, keepDev) {
  /* 清理闲置超过 180 天的槽（刚写的除外；只剩一个槽时不清） */
  const now = Date.now();
  const slots = d.slots || {};
  const stale = Object.keys(slots).filter(dev =>
    dev !== keepDev && now - (slots[dev].ts || 0) > SLOT_IDLE_PRUNE_MS);
  if (stale.length && Object.keys(slots).length - stale.length >= 1) {
    stale.forEach(dev => { delete slots[dev]; });
    log(`已清理 ${stale.length} 个闲置数据槽：${stale.join(', ')}`);
  }
}
function writeSlot(payload, dev, baseRev) {
  /* 槽级乐观锁：baseRev 与该槽当前 rev 不一致 → 409；各设备写各的槽互不冲突 */
  const d = readData();
  const slots = d.slots = d.slots || {};
  const cur = slots[dev] || { rev: 0, ts: 0, payload: null };
  if (baseRev != null && cur.rev !== baseRev) {
    return { ok: false, conflict: true, rev: d.rev || 0, slotRev: cur.rev || 0, payload: cur.payload };
  }
  d.rev = (d.rev || 0) + 1;
  slots[dev] = { rev: d.rev, ts: Date.now(), payload };
  pruneSlots(d, dev);
  writeFile(d);
  return { ok: true, rev: d.rev, ts: d.ts, slotRev: slots[dev].rev };
}
function writeLegacy(payload, baseRev) {
  /* 兼容 R23 旧客户端：整包替换 */
  const d = readData();
  if (baseRev != null && (d.rev || 0) !== baseRev) {
    return { ok: false, conflict: true, rev: d.rev || 0, slots: d.slots };
  }
  d.rev = (d.rev || 0) + 1;
  d.slots = { legacy: { rev: d.rev, ts: Date.now(), payload } };
  writeFile(d);
  return { ok: true, rev: d.rev, ts: d.ts };
}

function sendJSON(res, obj, code) {
  const body = Buffer.from(JSON.stringify(obj), 'utf8');
  res.writeHead(code || 200, {
    'Content-Type': 'application/json; charset=utf-8',
    'Content-Length': body.length,
    'Cache-Control': 'no-store',
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Headers': 'Content-Type',
    'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
  });
  res.end(body);
}
function serveIndex(req, res) {
  const file = fs.existsSync(INDEX) ? INDEX : findHtml();
  if (!file) { res.writeHead(404); return res.end('index.html 不存在'); }
  const body = fs.readFileSync(file);
  res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8', 'Content-Length': body.length, 'Cache-Control': 'no-store' });
  res.end(body);
}
function findHtml() {
  const f = fs.readdirSync(BASE_DIR).filter(x => /\.html$/i.test(x) && !/^校园网部署说明/.test(x));
  return f.length ? path.join(BASE_DIR, f[0]) : null;
}

const server = http.createServer((req, res) => {
  const u = new URL(req.url, 'http://localhost');
  const q = u.searchParams;
  const check = () => q.get('token') === TOKEN;

  if (req.method === 'OPTIONS') return sendJSON(res, { ok: true });

  if (u.pathname === '/api/ping') {
    return sendJSON(res, { ok: true, name: 'qiwu7', rev: readData().rev, time: new Date().toLocaleString('zh-CN') });
  }

  if (u.pathname === '/api/sync' && req.method === 'GET') {
    if (!check()) return sendJSON(res, { ok: false, error: '口令不对' }, 401);
    const d = readData();
    const slots = d.slots || {};
    /* 客户端带 revs（各槽已知版本，JSON）→ 只回有变化的槽，省带宽 */
    let revs = {};
    try { revs = JSON.parse(q.get('revs') || '{}'); if (!revs || typeof revs !== 'object') revs = {}; } catch (e) { revs = {}; }
    const changed = {};
    Object.keys(slots).forEach(dev => {
      if (!(dev in revs && revs[dev] === slots[dev].rev)) changed[dev] = slots[dev];
    });
    const slotsRev = {};
    Object.keys(slots).forEach(dev => { slotsRev[dev] = slots[dev].rev || 0; });
    return sendJSON(res, { ok: true, rev: d.rev || 0, ts: d.ts || 0, slots: changed, count: Object.keys(slots).length, slotsRev });
  }

  if (u.pathname === '/api/backup') {
    if (!check()) return sendJSON(res, { ok: false, error: '口令不对' }, 401);
    const body = Buffer.from(JSON.stringify(readData()), 'utf8');
    res.writeHead(200, {
      'Content-Type': 'application/json; charset=utf-8',
      'Content-Length': body.length,
      /* HTTP 头只能装 ASCII：中文文件名必须百分号编码（RFC 5987） */
      'Content-Disposition': `attachment; filename*=UTF-8''${encodeURIComponent('栖梧7班数据备份.json')}`,
    });
    return res.end(body);
  }

  if (u.pathname === '/api/sync' && req.method === 'POST') {
    if (!check()) return sendJSON(res, { ok: false, error: '口令不对' }, 401);
    let raw = '';
    let tooBig = false;
    req.on('data', c => { raw += c; if (raw.length > 50 * 1024 * 1024) { tooBig = true; req.destroy(); } });
    req.on('end', () => {
      if (tooBig) return;
      let reqObj = {};
      try { reqObj = raw ? JSON.parse(raw) : {}; } catch (e) {
        return sendJSON(res, { ok: false, error: '请求不是合法 JSON：' + e.message }, 400);
      }
      const payload = reqObj.payload;
      if (payload === undefined || payload === null) {
        return sendJSON(res, { ok: false, error: '缺少 payload' }, 400);
      }
      const baseRev = reqObj.baseRev;
      let r;
      let dev = null;
      if (reqObj.deviceId !== undefined && reqObj.deviceId !== null) {
        /* R24 新协议：写自己的数据槽 */
        dev = String(reqObj.deviceId).slice(0, 64) || 'unknown';
        r = writeSlot(payload, dev, baseRev);
      } else {
        /* 旧协议：整包替换（兼容升级过渡期） */
        r = writeLegacy(payload, baseRev);
      }
      if (!r.ok) {
        return sendJSON(res, {
          ok: false, conflict: true, rev: r.rev || 0, slotRev: r.slotRev != null ? r.slotRev : undefined,
          payload: r.payload, slots: r.slots,
          error: '数据已被其他设备更新，请合并后重试',
        }, 409);
      }
      log(`已同步 rev=${r.rev}（${dev || 'legacy'}，来自 ${req.socket.remoteAddress}）`);
      return sendJSON(res, { ok: true, rev: r.rev, ts: r.ts, slotRev: r.slotRev });
    });
    return;
  }

  if (req.method === 'GET') return serveIndex(req, res);
  sendJSON(res, { ok: false, error: 'not found' }, 404);
});

0 · 一句话：这是个什么东西

这是一个班级管理网页（班主任用的：学生名单、加减分、考勤、成绩分析、表扬墙、奖品兑换……）。 现在要把它放到校园网里，让老师的办公电脑、课室的一体机、学生登录的个人端都能打开同一个网址， 并且数据实时同步——谁改了什么，其他端几十秒内就能看到。

技术形态很简单：一个 HTML 网页 + 一个只有一百多行的同步服务（Python 或 Node 二选一， 不需要装任何第三方库、不需要数据库、不需要外网、不花钱）。数据就存成服务器上的一个 JSON 文件。

1 · 需要什么（很低的门槛）

项目要求
一台机器校园网内一台常开的电脑或服务器。办公机、机房机器都行。 配置无要求（1 核 1G 都够）。关机 = 大家同步不了（但已打开的设备仍能离线用，恢复后自动补同步）。
运行环境Python 3.6+（推荐）或 Node 12+，有一个就行。Linux 一般自带 Python3。
一个端口默认 8080。被占用就换（9000、8888 都行）。校园网内互通即可。
磁盘几十 MB 足够（含自动备份）。
不需要不需要域名、不需要公网 IP、不需要备案、不需要数据库、不需要外网访问。

4 · 常见问题

现象怎么办
提示「端口被占用」换端口：Windows 改 bat 里的 QIWU7_PORT；Linux 用 QIWU7_PORT=9000 ./启动_Linux.sh。也可以命令行直接 python qiwu7_server.py 9000。
本机打得开，别的电脑打不开① 防火墙没放行（见第 2 步第 3 点）② 两台机器不在同一个网段/VLAN ③ 地址里的 IP 写错了（要用 192.168.x.x 那种局域网 IP，不是 127.0.0.1）。
没有 Python 也没有 Node装 Python 最省事：python.org 下载，安装时勾选 "Add Python to PATH"。 Linux：sudo apt install -y python3。
电脑重启后就没了需要开机自启。Windows：把 启动_Windows.bat 的快捷方式放进 shell:startup 目录。Linux：写一个 systemd 服务（下面是模板）。
数据存在哪qiwu7_data/data.json。每次写入前自动备份到 qiwu7_data/backups/，保留最近 10 份。可以直接拷贝整个文件夹做冷备份。
数据文件损坏了程序会自动回退到最近一份备份。也可以手动从 backups/ 里挑一份改名覆盖 data.json。
想看服务器有没有在跑浏览器打开 http://服务器IP:8080/api/ping，返回 {"ok":true,"rev":N} 就正常，rev 是数据版本，每次同步 +1。
多人同时改会不会乱不会。服务端有乐观锁：如果两台设备同时改，后提交的会收到冲突提示， 客户端自动重新合并再提交。数据包本身按唯一编号合并，重复提交不会重复计分。
服务器关了会怎样所有端自动降级为本地模式，功能照常用（数据存在各自浏览器里）； 服务器恢复后自动接着同步，不会丢。

6 · 安全提醒

① 一定要改掉默认口令 qiwu7 —— 这个口令是唯一的一道门，虽然只在校园网内。
　　不改会怎样？qiwu7 是公开写在启动脚本和本文档里的，等于没锁。校园网内任何知道这台机器地址的人，
　　都能直接下载全班数据（成绩 / 名单 / 记录），也能提交假数据覆盖整库。服务启动时如果还是默认口令，会打印醒目警告。
　　⚠ 改完口令，教师端「设置 / 数据 → 🌐 校园网同步」里的口令也要跟着改，否则会提示「口令不对」。
② 只在校园网内放通端口，不要在路由器上做端口映射暴露到公网。
③ 建议每周把 qiwu7_data 整个文件夹复制一份到网盘或移动硬盘（里面有自动备份）。
④ 数据含学生姓名与成绩，属敏感信息，请按学校规定保管；不用时直接关掉服务即可。
